/** Catégories du catalogue (E4 : when exhaustif sur enum). */
enum class Categorie { ALIMENTAIRE, HYGIENE, LOISIRS }

/** États du panier (E4 : when exhaustif sur sealed). */
sealed interface EtatPanier {
    data object Vide : EtatPanier
    data class EnCours(val articles: List<Article>) : EtatPanier
    data class Valide(val totalCentimes: Int) : EtatPanier
}

/** Entité du catalogue. */
data class Article(
    val id: Long,
    val titre: String,
    val prixCentimes: Int,
    val categorie: Categorie,
)

/** Accès aux données — les lectures sont SUSPEND (E7). */
interface DepotArticle {
    /** Charge tout le catalogue (réseau simulé). */
    suspend fun chargerTous(): List<Article>

    /** Recherche par préfixe de titre. */
    suspend fun chercher(prefixe: String): List<Article>
}

/** Implémentation mémoire du dépôt (données fictives). */
class DepotArticleMemoire : DepotArticle {
    private val catalogue = listOf(
        Article(id = 1, titre = "Lait demi-écrémé", prixCentimes = 109, categorie = Categorie.ALIMENTAIRE),
        Article(id = 2, titre = "Savon doux", prixCentimes = 249, categorie = Categorie.HYGIENE),
        Article(id = 3, titre = "Jeu de cartes", prixCentimes = 599, categorie = Categorie.LOISIRS),
    )

    override suspend fun chargerTous(): List<Article> =
        withContext(Dispatchers.IO) {
            delay(50) // réseau simulé
            catalogue
        }

    override suspend fun chercher(prefixe: String): List<Article> =
        chargerTous().filter { it.titre.startsWith(prefixe, ignoreCase = true) }
}

/** ViewModel minimaliste (sans dépendance lifecycle — seul le contrat compte). */
class PanierViewModel(private val depot: DepotArticle) {
    var etat: EtatPanier = EtatPanier.Vide
        private set

    /** Recharge le catalogue — appelable DEPUIS une coroutine (contraste E7). */
    suspend fun rafraichir() {
        etat = EtatPanier.EnCours(depot.chargerTous())
    }
}

/** Contrat d'écran (E5 : stubs d'override). */
interface Ecran {
    val titreEcran: String

    fun actualiser()
}
